1. 获取可用打卡地点

// GET /checkin-location/available-locations
// 请求体：无
// 响应体解释：

// longitude/latitude：GCJ-02 坐标，示例为脱敏占位
// radiusMeter：打卡有效半径，单位米

  "code": 0, // 业务状态码，0 成功
  "msg": "操作成功", // 提示信息
  "data": [
    {
      "id": "1", // 地点 ID
      "createTime": "2026-09-08 12:00:00", // 创建时间
      "updateTime": "2026-10-20 11:00:00", // 更新时间
      "locationName": "某组团", // 地点名称
      "address": "", // 地址，可为空字符串
      "longitude": 104.000000, // 经度，GCJ-02
      "latitude": 30.600000, // 纬度，GCJ-02
      "radiusMeter": 140, // 打卡半径，米

5. 学生提交打卡

请求体

// POST /student-checkin-record/checkin
// 请求头：Content-Type: application/json；Cookie: <登录态Cookie>
// 请求体解释：
// latitude/longitude：当前坐标，GCJ-02
// reverseGeocode：逆地理地址
// clientTime：客户端时间
// userAgent：客户端 UA
// coordinateSystemCode：坐标系编码
// accuracy：定位精度，米
// locationDurationMs：定位耗时，毫秒
// networkTypeCode：网络类型
// locationEnabled：定位开关状态
// locationAuthorized：定位授权信息
// locationAccuracy：定位精度描述
// errorMessage：定位错误信息
{
  "latitude": 30.600100, // 当前纬度，GCJ-02
  "longitude": 104.000100, // 当前经度，GCJ-02
  "reverseGeocode": "成都市某区某路", // 逆地理地址
  "clientTime": "2026-10-21 21:00:00", // 客户端时间
  "userAgent": "Mozilla/5.0", // 客户端 UA
  "coordinateSystemCode": "gcj02", // 坐标系
  "accuracy": 10, // 定位精度，米
  "locationDurationMs": 500, // 定位耗时，毫秒
  "networkTypeCode": "4g", // 网络类型
  "locationEnabled": false, // 定位开关状态
  "locationAuthorized": "", // 定位授权信息
  "locationAccuracy": "", // 定位精度描述
  "errorMessage": "" // 定位错误信息
}
​

响应体

// 响应体解释：
// code：0 成功
// data：true 表示打卡成功
// ok：请求是否成功
{
  "code": 0, // 0 成功
  "msg": "操作成功", // 提示信息
  "data": true, // 打卡结果，true 成功
  "ok": true // 是否成功
}
